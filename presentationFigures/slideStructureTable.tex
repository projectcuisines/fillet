\rowcolor{rowtint}
\textbf{AVALON}\newline{\tiny Julia}
 & one $T$; land/ocean blended
 & 89 equal-area in $\sin\phi$, nodes at edges$^\dagger$
 & linear $A+BT$; $A{=}210$, $B{=}2.0$; Myhre CO$_2$ forcing
 & \textbf{no atmosphere}: prescribed surface albedo \emph{is} planetary; no zenith term
 & hard step at \code{T\_ice}
 & outermost band below \code{T\_ice}, snapped; $-10\degr$C$^\dagger$
 & constant $D$, as prescribed
 & none
 & IMEX, $\Delta t\simeq1$ month \\

\textbf{HEXTOR}\newline{\tiny Fortran}
 & one $T$; ocean fraction per belt
 & 18 equal-width $10\degr$ belts
 & RT lookup table, trilinear in $(\log p,\log X_{\rm CO_2},T)$
 & RT lookup of planetary albedo; \textbf{4 zenith nodes}; \code{fcloud}$=0$
 & \textbf{ramp}, $10$~K window
 & $263.15$~K crossing, interpolated
 & \textbf{rescaled at run time}; submission files $d_0$, $10\%$ low
 & carbon--silicate cycle (off), CO$_2$ condensation
 & explicit, $\Delta t = 4.3\times10^{4}$~s \\

\rowcolor{rowtint}
\textbf{VPLanet}\newline{\tiny C (POISE)}
 & \textbf{two} $T$ (land, water), coupled
 & 150 equal-area in $\sin\phi$
 & fixed $A+BT$ for Ben\,1--3, Exp\,1--3$^\dagger$; WK97 for Exp\,4$^\dagger$
 & surface albedo $+\;0.08(3\sin^2 z-1)/2$ zenith term
 & hard step at $-2\degr$C
 & $273.15$~K land, $271.15$~K ocean, snapped
 & constant $D$, as prescribed
 & \textbf{ice sheets with flow} (on); sea ice (off)
 & seasonal, 60 steps/orbit \\

\textbf{OPS}\newline{\tiny Fortran}
 & one $T$; ocean fraction per belt
 & 36 equal-width $5\degr$ belts
 & RT lookup table, bilinear in $(T,\ln p_{\rm CO_2})$; \emph{extrapolates}
 & RT lookup; \textbf{19 zenith nodes}; Fresnel ocean; cloud diagnosed per belt
 & \textbf{ramp}, $34$~K window; snow albedo $T$- and star-dependent
 & $263.15$~K crossing, interpolated
 & rescaled \emph{every step}; a function of the climate state
 & weathering, CO$_2$ cloud, diagnosed H$_2$O cloud
 & explicit, $\Delta t = 8.6\times10^{3}$~s \\

\rowcolor{rowtint}
\textbf{Shields-}\newline\textbf{Bitz}\newline{\tiny Python}
 & \textbf{two} $T$ (land, water), $\nu{=}3.0$
 & \sbBoxes\ equal-area in $\sin\phi$
 & linear $A+BT$; $A{=}203.3$, $B{=}2.09$; \textbf{no CO$_2$ term at all}
 & \textbf{no atmosphere}; albedos are broadband values weighted for the host star
 & hard step at $-2\degr$C, but only $0.18$ tall
 & seasonal mean of daily \emph{ocean} freezing latitude, on the wrong axis
 & Hadley-cell \textbf{profile} $D(x)$, not a constant
 & \textbf{thermodynamic sea ice} (thickness, latent heat)
 & implicit, 360 steps/orbit \\
